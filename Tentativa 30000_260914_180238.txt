/* =================================
   PAGE NAVIGATION
================================= */

function showPage(pageName) {

  document
    .querySelectorAll(".page")
    .forEach(page => {
      page.classList.remove("active");
    });

  const target =
    document.getElementById(
      "page-" + pageName
    );

  if (target) {
    target.classList.add("active");
  }

  document
    .querySelectorAll(".nav-button")
    .forEach(button => {

      button.classList.toggle(
        "active",
        button.dataset.page === pageName
      );

    });

  const titles = {

    home: "ILLIT Archive",

    profile: "Profile",

    calendar: "Calendar",

    activities: "Activities",

    status: "Status"

  };

  document.getElementById(
    "headerTitle"
  ).textContent =
    titles[pageName] || "ILLIT Archive";

  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });

  updateFloatingButton(pageName);
}

/* =================================
   FLOATING BUTTON
================================= */

function updateFloatingButton(pageName) {

  const button =
    document.getElementById(
      "floatingAdd"
    );

  if (
    pageName === "calendar" ||
    pageName === "activities"
  ) {

    button.style.display = "flex";

  } else {

    button.style.display = "none";

  }

}

function handleFloatingAdd() {

  const active =
    document.querySelector(
      ".page.active"
    );

  if (!active) return;

  if (
    active.id === "page-calendar"
  ) {

    if (typeof openCalendarEventModal === "function") {
      openCalendarEventModal();
    }

    return;
  }

  if (
    active.id === "page-activities"
  ) {

    if (typeof openActivityModal === "function") {
      openActivityModal();
    }

  }

}

/* =================================
   MODAL
================================= */

function openModal(content) {

  const backdrop =
    document.getElementById(
      "modalBackdrop"
    );

  const modal =
    document.getElementById(
      "modalContent"
    );

  modal.innerHTML = content;

  backdrop.classList.add("show");

  document.body.style.overflow = "hidden";
}

function closeModal() {

  const backdrop =
    document.getElementById(
      "modalBackdrop"
    );

  backdrop.classList.remove("show");

  document.body.style.overflow = "";

}

function closeModalFromBackdrop(event) {

  if (
    event.target.id ===
    "modalBackdrop"
  ) {

    closeModal();

  }

}

/* =================================
   TOAST
================================= */

function showToast(message) {

  const toast =
    document.getElementById(
      "toast"
    );

  toast.textContent = message;

  toast.classList.add("show");

  setTimeout(() => {

    toast.classList.remove("show");

  }, 2200);

}

/* =================================
   SETTINGS
================================= */

function openSettings() {

  showToast(
    "Settings will be available here."
  );

}

/* =================================
   GET MEMBERS
================================= */

function getMembers() {

  const saved =
    localStorage.getItem(
      MEMBER_STORAGE
    );

  if (!saved) {

    localStorage.setItem(
      MEMBER_STORAGE,
      JSON.stringify(DEFAULT_MEMBERS)
    );

    return DEFAULT_MEMBERS;

  }

  try {

    const parsed =
      JSON.parse(saved);

    if (
      Array.isArray(parsed) &&
      parsed.length === 5
    ) {

      return parsed;

    }

  } catch (error) {

    console.log(error);

  }

  return DEFAULT_MEMBERS;

}

/* =================================
   SAVE MEMBERS
================================= */

function saveMembers(members) {

  localStorage.setItem(
    MEMBER_STORAGE,
    JSON.stringify(members)
  );

}

/* =================================
   ESCAPE HTML
================================= */

function escapeHTML(value) {

  if (value === undefined ||
      value === null) {

    return "";

  }

  return String(value)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");

}

/* =================================
   PHOTO UPLOAD
================================= */

function handleProfilePhoto(event) {

  const file =
    event.target.files[0];

  if (!file) return;

  if (!file.type.startsWith("image/")) {

    showToast(
      "Please select an image."
    );

    return;

  }

  const reader =
    new FileReader();

  reader.onload = function(e) {

    editingPhotoData =
      e.target.result;

    const preview =
      document.getElementById(
        "profileEditorPreview"
      );

    if (!preview) return;

    if (preview.tagName === "IMG") {

      preview.src =
        editingPhotoData;

    } else {

      preview.outerHTML = `

        <img
          id="profileEditorPreview"
          src="${editingPhotoData}"
          alt="Profile photo"
        >

      `;

    }

  };

  reader.readAsDataURL(file);

}

function addProfileEditButtons() {

  const cards =
    document.querySelectorAll(
      ".member-card"
    );

  cards.forEach(card => {

    const memberId =
      card.dataset.memberId;

    const back =
      card.querySelector(
        ".member-card-back"
      );

    if (!back) return;

    if (
      back.querySelector(
        ".edit-profile-button"
      )
    ) return;

    const button =
      document.createElement(
        "button"
      );

    button.className =
      "edit-profile-button";

    const member =
      getMembers().find(
        item =>
          item.id === memberId
      );

    button.style.background =
      member
        ? member.color
        : "#17171a";

    button.textContent =
      "Edit Profile";

    button.onclick =
      function(event) {

        event.stopPropagation();

        openProfileEditor(
          memberId
        );

      };

    const openButton =
      back.querySelector(
        ".open-profile-button"
      );

    if (openButton) {

      openButton.insertAdjacentElement(
        "afterend",
        button
      );

    } else {

      back.appendChild(button);

    }

  });

}

function calendarDateKey(date){
  const year = date.getFullYear();
  const month = String(
    date.getMonth() + 1
  ).padStart(2,"0");

  const day = String(
    date.getDate()
  ).padStart(2,"0");

  return `${year}-${month}-${day}`;
}

  const date = new Date(

      "en-US",
      {weekday:"long"}
    ),

    month: date.toLocaleDateString(

function saveCalendarEvent(){

  const date =
    document.getElementById(
      "calendarEventDate"
    ).value;

  const member =
    document.getElementById(
      "calendarEventMember"
    ).value;

  const title =
    document.getElementById(
      "calendarEventTitle"
    ).value.trim();

  const time =
    document.getElementById(
      "calendarEventTime"
    ).value.trim();

  const description =
    document.getElementById(
      "calendarEventDescription"
    ).value.trim();

  if(!date){

    showToast("Please choose a date.");

    return;
  }

  if(!title){

    showToast("Please add an event name.");

    return;
  }

  const events = getCalendarEvents();

  if(editingCalendarEventId){

    const index = events.findIndex(
      item =>
        item.id === editingCalendarEventId
    );

    if(index !== -1){

      events[index] = {
        ...events[index],
        date,
        member,
        title,
        time,
        description
      };

    }

  }else{

    events.push({

      id:
        "event_" +
        Date.now() +
        "_" +
        Math.random()
          .toString(36)
          .slice(2),

      date,
      member,
      title,
      time,
      description

    });

  }

  saveCalendarEvents(events);

  editingCalendarEventId = null;

  closeModal();

  renderCalendar();

  showToast("Event saved.");
}

}

function renderCalendarEvents(){

  const events = getCalendarEvents();

  const year =
    calendarCurrentDate.getFullYear();

  const month =
    calendarCurrentDate.getMonth();

  const monthEvents =
    events
      .filter(event => {

        if(event.member !== calendarActiveTab){
          return false;
        }

        const eventDate =
          new Date(event.date + "T12:00:00");

        return (
          eventDate.getFullYear() === year &&
          eventDate.getMonth() === month
        );

      })
      .sort((a,b) =>
        a.date.localeCompare(b.date)
      );

  if(monthEvents.length === 0){

    return `

      <div class="calendar-empty">

        <span class="calendar-empty-star">
          ✦
        </span>

        No activities for this month yet.

      </div>

    `;

  }

  const grouped = {};

  monthEvents.forEach(event => {

    if(!grouped[event.date]){
      grouped[event.date] = [];
    }

    grouped[event.date].push(event);

  });

  return `

    <div class="calendar-event-list">

      ${
        Object.keys(grouped)
          .sort()
          .map(date => {

            const dateInfo =
              formatCalendarDate(date);

            return `

              <div class="calendar-day-card">

                <div class="calendar-day-header">

                  <div>
                    <div class="calendar-day-number">
                      ${dateInfo.day}
                    </div>

                    <div class="calendar-day-name">
                      ${dateInfo.weekday}
                    </div>
                  </div>

                  <div class="calendar-day-name">
                    ${dateInfo.month}
                  </div>

                </div>

                ${
                  grouped[date]
                    .map(event => {

                      const memberInfo =
                        CALENDAR_MEMBER_DATA[
                          event.member
                        ] ||
                        CALENDAR_MEMBER_DATA.illit;

                      return `

                        <div class="calendar-event">

                          <div
                            class="calendar-star"
                            style="color:${memberInfo.color}"
                          >
                            ★
                          </div>

                          <div class="calendar-event-content">

                            <div class="calendar-event-title">
                              ${escapeHTML(event.title)}
                            </div>

                            ${
                              event.time
                              ? `
                                <div class="calendar-event-meta">
                                  ${escapeHTML(event.time)}
                                </div>
                              `
                              : ""
                            }

                            ${
                              event.description
                              ? `
                                <div class="calendar-event-meta">
                                  ${escapeHTML(event.description)}
                                </div>
                              `
                              : ""
                            }

                          </div>

                          <div class="calendar-event-actions">

                            <button
                              class="calendar-event-action"
                              onclick="openCalendarEventEditor('${event.id}')"
                            >
                              ✎
                            </button>

                          </div>

                        </div>

                      `;

                    })
                    .join("")

                }

              </div>

            `;

          })
          .join("")

      }

    </div>

  `;
}

        ...activities[index],

        member,
        type,
        title,
        date,
        link,
        description

      };

    }

  }else{

    activities.push({

      id:
        "activity_" +
        Date.now() +
        "_" +
        Math.random()
          .toString(36)
          .slice(2),

      member,
      type,
      title,
      date,
      link,
      description

    });

  }

  saveActivities(activities);

  editingActivityId = null;

  closeModal();

  renderActivities();

  showToast(
    "Activity saved."
  );

}

/* =========================
   GET MONTHS
========================= */

function getStatusMonths(){

  const activities =
    getActivities();

  const current =
    new Date();

  const months = [];

  let earliest = null;

  activities.forEach(activity => {

    if(!activity.date){
      return;
    }

    const date =
      new Date(
        activity.date + "T12:00:00"
      );

    if(
      !earliest ||
      date < earliest
    ){

      earliest = date;

    }

  });

  if(!earliest){

    return [

      {
        key:
          `${current.getFullYear()}-${String(
            current.getMonth()+1
          ).padStart(2,"0")}`,

        label:
          current.toLocaleDateString(
            "en-US",
            {
              month:"short",
              year:"numeric"
            }
          )
      }

    ];

  }

  const start =
    new Date(
      earliest.getFullYear(),
      earliest.getMonth(),
      1
    );

  const end =
    new Date(
      current.getFullYear(),
      current.getMonth(),
      1
    );

  let cursor =
    new Date(start);

  while(cursor <= end){

    const year =
      cursor.getFullYear();

    const month =
      cursor.getMonth() + 1;

    const key =
      `${year}-${String(month).padStart(2,"0")}`;

    const label =
      cursor.toLocaleDateString(
        "en-US",
        {
          month:"short",
          year:"numeric"
        }
      );

    months.push({
      key,
      label
    });

    cursor =
      new Date(
        year,
        cursor.getMonth()+1,
        1
      );

  }

  return months;

}

/* =========================
   COUNT ACTIVITIES
========================= */

function getStatusData(){

  const activities =
    getActivities();

  const months =
    getStatusMonths();

  return months.map(month => {

    const row = {
      key:month.key,
      label:month.label
    };

    Object.keys(
      STATUS_MEMBER_DATA
    ).forEach(member => {

      row[member] =
        activities.filter(
          activity => {

            if(
              activity.member !== member
            ){
              return false;
            }

            if(!activity.date){
              return false;
            }

            return activity.date
              .startsWith(
                month.key
              );

          }
        ).length;

    });

    return row;

  });

}

/* =========================
   TOTAL MAX
========================= */

function getStatusMax(data){

  let max = 0;

  data.forEach(row => {

    Object.keys(
      STATUS_MEMBER_DATA
    ).forEach(member => {

      max =
        Math.max(
          max,
          row[member] || 0
        );

    });

  });

  if(max === 0){
    return 1;
  }

  return max;

}

/* =========================
   SVG PATH
========================= */

function createStatusPath(
  data,
  member,
  max
){

  if(data.length === 0){
    return "";
  }

  const width = 100;
  const height = 100;

  const points =
    data.map(
      (row,index) => {

        const x =
          data.length === 1
          ? 50
          : (
            index /
            (data.length-1)
          ) * width;

        const value =
          row[member] || 0;

        const y =
          height -
          (
            value /
            max
          ) * height;

        return `${x},${y}`;

      }
    );

  return points.join(" ");

}

function importArchiveData(event){

    event.target.files[0];

    new FileReader();

  reader.onload = function(){

    try{

      const backup =
        JSON.parse(
          reader.result
        );

      if(
        backup.members !== undefined
      ){

        if(
          backup.members === null
        ){

          localStorage.removeItem(
            "illit_archive_members"
          );

        }else{

          localStorage.setItem(
            "illit_archive_members",
            backup.members
          );

        }

      }

      if(
        backup.calendar !== undefined
      ){

        if(
          backup.calendar === null
        ){

          localStorage.removeItem(
            "illit_archive_calendar"
          );

        }else{

          localStorage.setItem(
            "illit_archive_calendar",
            backup.calendar
          );

        }

      }

      if(
        backup.activities !== undefined
      ){

        if(
          backup.activities === null
        ){

          localStorage.removeItem(
            "illit_archive_activities"
          );

        }else{

          localStorage.setItem(
            "illit_archive_activities",
            backup.activities
          );

        }

      }

      if(
        backup.logo !== undefined
      ){

        if(
          backup.logo === null
        ){

          localStorage.removeItem(
            SETTINGS_LOGO_STORAGE
          );

        }else{

          localStorage.setItem(
            SETTINGS_LOGO_STORAGE,
            backup.logo
          );

        }

      }

      renderProfileCards();

      renderCalendar();

      renderActivities();

      renderStatusGraph();

      updateSiteLogo();

      closeModal();

      showToast(
        "Backup imported."
      );

    }catch(error){

      showToast(
        "Invalid backup file."
      );

    }

  };

  reader.readAsText(file);

}

   RESET

    "illit_archive_calendar"
  );

  localStorage.removeItem(
    "illit_archive_activities"
  );

  localStorage.removeItem(
    SETTINGS_LOGO_STORAGE
  );

  renderProfileCards();

  renderCalendar();

  renderActivities();

  renderStatusGraph();

  updateSiteLogo();

  closeModal();

  showToast(
    "Archive data reset."
  );

}

const oldHandleProfilePhoto =

window.handleProfilePhoto =
  function(event){

    const file =
      event.target.files[0];

    if(!file){
      return;
    }

    const reader =
      new FileReader();

    reader.onload =
      function(){

        editingPhotoData =
          reader.result;

        const preview =
          document.getElementById(
            "profileEditorPreview"
          );

        const empty =
          document.getElementById(
            "profileEditorEmpty"
          );

        if(preview){

          preview.src =
            editingPhotoData;

          preview.style.display =
            "block";

        }

        if(empty){

          empty.style.display =
            "none";

        }

      };

    reader.readAsDataURL(file);

  };

const oldRenderCalendar =
  window.renderCalendar;

window.renderCalendar =
  function(){

    const container =
      document.getElementById(
        "calendarContainer"
      );

    if(!container){
      return;
    }

    const events =
      getCalendarEvents();

    const year =
      calendarCurrentDate.getFullYear();

    const month =
      calendarCurrentDate.getMonth();

    const filtered =
      events.filter(
        event => {

          const date =
            new Date(
              event.date +
              "T00:00:00"
            );

          const sameMonth =
            date.getFullYear() === year &&
            date.getMonth() === month;

          const sameMember =
            calendarActiveTab === "illit"
              ? event.member === "illit"
              : event.member === calendarActiveTab;

          return (
            sameMonth &&
            sameMember
          );

        }
      )
      .sort(
        (a,b) =>
          a.date.localeCompare(
            b.date
          )
      );

    let html = "";

    html +=
      renderCalendarHeader();

    html +=
      renderCalendarTabs();

    html += `

      <button
        class="calendar-add-button"
        onclick="openCalendarAdd()"
      >
        + Add Event
      </button>

    `;

    if(!filtered.length){

      html += `

        <div class="calendar-empty">

          <div class="calendar-empty-star">
            ✦
          </div>

          <div class="calendar-empty-title">
            No events yet
          </div>

          <div class="calendar-empty-text">
            Add an event to this month.
          </div>

        </div>

      `;

      container.innerHTML =
        html;

      return;

    }

    html +=
      `<div class="calendar-event-list">`;

    let previousDate = "";

    filtered.forEach(
      event => {

        if(
          event.date !== previousDate
        ){

          const date =
            new Date(
              event.date +
              "T00:00:00"
            );

          html += `

            <div class="calendar-day">

              ${date.toLocaleDateString(
                "en-US",
                {
                  weekday:"long",
                  month:"short",
                  day:"numeric"
                }
              )}

            </div>

          `;

          previousDate =
            event.date;

        }

        const starColor =
          calendarStarColor(
            event.member
          );

        html += `

          <div class="calendar-event">

            <div
              class="calendar-event-star"
              style="
                color:${starColor}
              "
            >
              ★
            </div>

            <div class="calendar-event-content">

              <div class="calendar-event-title">

                ${escapeHTML(
                  event.title || ""
                )}

              </div>

              ${
                event.time

                ?

                `
                <div class="calendar-event-time">
                  ${escapeHTML(event.time)}
                </div>
                `

                :

                ""
              }

              ${
                event.description

                ?

                `
                <div class="calendar-event-description">
                  ${escapeHTML(
                    event.description
                  )}
                </div>
                `

                :

                ""
              }

            </div>

            <div class="calendar-event-actions">

              <button
                class="calendar-event-action"
                onclick="
                  editCalendarEvent(
                    '${event.id}'
                  )
                "

                ✎

                ×
              </button>

            </div>

          </div>

        `;

      }
    );

      `</div>`;

      html;

  };

function statusMonthLabel(
  year,
  month
){

  const date =
    new Date(
      year,
      month,
      1
    );

  return date.toLocaleDateString(
    "en-US",
    {
      month:"short"
    }
  );

}

/* =========================
   MONTH KEY
========================= */

function statusMonthKey(
  year,
  month
){

  return (
    year +
    "-" +
    String(
      month + 1
    ).padStart(
      2,
      "0"
    )
  );

}

/* =========================
   GET ACTIVITY MONTH
========================= */

function getActivityMonth(
  activity
){

  if(!activity.date){
    return null;
  }

  const parts =
    activity.date.split("-");

  if(
    parts.length < 2
  ){
    return null;
  }

  return (
    parts[0] +
    "-" +
    parts[1]
  );

}

/* =========================
   BUILD STATUS MONTHS
========================= */

function buildStatusMonths(){

  const activities =
    getActivities();

  const dates =
    activities

      .map(
        activity =>
          activity.date
      )

      .filter(Boolean);

  const now =
    new Date();

  let start;

  if(dates.length){

    const sorted =
      [...dates].sort();

    const first =
      sorted[0].split("-");

    start =
      new Date(
        Number(first[0]),
        Number(first[1]) - 1,
        1
      );

  }else{

    start =
      new Date(
        now.getFullYear(),
        now.getMonth(),
        1
      );

  }

  const end =
    new Date(
      now.getFullYear(),
      now.getMonth(),
      1
    );

  const months = [];

  const cursor =
    new Date(
      start
    );

  while(
    cursor <= end
  ){

    months.push({

      year:
        cursor.getFullYear(),

      month:
        cursor.getMonth(),

      key:
        statusMonthKey(
          cursor.getFullYear(),
          cursor.getMonth()
        )

    });

    cursor.setMonth(
      cursor.getMonth() + 1
    );

  }

  return months;

}

  const activities =
    getActivities();

  const months =
    buildStatusMonths();

  return STATUS_MEMBERS.map(
    member => {

      let cumulative = 0;

      const values =
        months.map(
          month => {

            const monthly =
              activities.filter(
                activity => {

                  return (
                    activity.member ===
                    member.id &&

                    getActivityMonth(
                      activity
                    ) ===
                    month.key
                  );

                }
              ).length;

            cumulative +=
              monthly;

            return {

              ...month,

              value:
                cumulative

            };

          }
        );

      return {

        ...member,

        values

      };

    }
  );

}

/* =========================
   GRAPH POINTS
========================= */

function statusPoints(
  values,
  maxValue,
  width,
  height
){

  if(!values.length){
    return "";
  }

  const safeMax =
    Math.max(
      maxValue,
      1
    );

  return values.map(
    (value,index) => {

      const x =
        values.length === 1

          ? width / 2

          :

          (
            index /
            (
              values.length - 1
            )
          ) * width;

      const y =
        height -
        (
          value /
          safeMax
        ) * height;

      return {
        x,
        y
      };

    }
  );

}

/* =========================
   SVG PATH
========================= */

function statusPath(
  points
){

  if(!points.length){
    return "";
  }

  return points.map(
    (point,index) => {

      return (
        index === 0
          ? `M ${point.x} ${point.y}`
          : `L ${point.x} ${point.y}`
      );

    }
  ).join(" ");

}

function renderStatus(){

  const container =
    document.getElementById(
      "statusContainer"
    );

  if(!container){
    return;
  }

    buildStatusData();

  const months =
    buildStatusMonths();

    getActivities();

    !activities.length
  ){

        <div class="status-intro">

          <div class="status-subtitle">

          </div>

          </div>

          <div class="status-empty-title">
            No activity data yet
          </div>

          <div class="status-empty-text">
            Add activities to start building the graph.
          </div>

        </div>

      </div>

    `;

    return;

  }

  const allValues =
    graphData.flatMap(
      member =>
        member.values.map(
          point =>
            point.value
        )
    );

  const maxValue =
    Math.max(
      ...allValues,
      1
    );

  const width =
    Math.max(
      500,
      months.length * 55
    );

  const height =
    205;

  let svg = "";

  graphData.forEach(
    member => {

      const points =
        statusPoints(
          member.values,
          maxValue,
          width,
          height
        );

      svg += `

        <polyline

          points="
            ${points.map(
              point =>
                `${point.x},${point.y}`
            ).join(" ")}
          "

          fill="none"

          stroke="${member.color}"

          stroke-width="2.5"

          stroke-linecap="round"

          stroke-linejoin="round"

        />

      `;

      points.forEach(
        point => {

          svg += `

            <circle

              cx="${point.x}"

              cy="${point.y}"

              r="3"

              fill="${member.color}"

            />

          `;

        }
      );

    }
  );

  const ySteps = 4;

  const yLabels =
    Array.from(
      {
        length:
          ySteps + 1
      },
      (_,index) => {

        const value =
          Math.round(
            maxValue -
            (
              index *
              (
                maxValue /
                ySteps
              )
            )
          );

        return value;

      }
    );

  container.innerHTML = `

    <div class="status-page">

      <div class="status-intro">

        <div class="status-subtitle">

      </div>

      <div class="status-graph-card">

        <div class="status-graph-header">

          <div>

            <div class="status-graph-heading">
              Activity evolution
            </div>

            <div class="status-graph-note">
              Cumulative activities
            </div>

          </div>

        </div>

        <div class="status-chart-wrapper">

          <div
            class="status-chart"
            style="width:${width + 42}px"
          >

            <div class="status-y-labels">

              ${
                yLabels.map(
                  value => `

                    <div class="status-y-label">
                      ${value}
                    </div>

                  `
                ).join("")
              }

            </div>

            <div class="status-chart-grid">

              ${
                yLabels.map(
                  () => `

                    <div
                      class="status-grid-line"
                    ></div>

                  `
                ).join("")
              }

            </div>

            <svg

              class="status-svg"

              viewBox="
                0
                0
                ${width}
                ${height}
              "

              preserveAspectRatio="
                none
              "

            >

              ${svg}

            </svg>

            <div class="status-x-labels">

              ${
                months.map(
                  month => `

                    <div class="status-x-label">

                      ${
                        statusMonthLabel(
                          month.year,
                          month.month
                        )
                      }

                      <br>

                      ${
                        month.year
                      }

                    </div>

                  `
                ).join("")
              }

            </div>

          </div>

        </div>

        <div class="status-legend">

          ${
            graphData.map(
              member => `

                <div class="status-legend-item">

                  <span
                    class="status-legend-dot"
                    style="
                      background:
                      ${member.color}
                    "
                  ></span>

                  ${member.name}

                </div>

              `
            ).join("")
          }

/* =========================
   PROCESS BACKUP
========================= */

function importArchiveData(
  event
){

  const file =
    event.target.files?.[0];

  if(!file){
    return;
  }

  const reader =
    new FileReader();

  reader.onload =
    function(e){

      try{

        const backup =
          JSON.parse(
            e.target.result
          );

        if(
          !backup ||
          !backup.data
        ){

          throw new Error(
            "Invalid backup"
          );

        }

        const data =
          backup.data;

        if(
          data.members !==
          undefined
        ){

          if(data.members === null){

            localStorage.removeItem(
              "illit_archive_members"
            );

          }else{

            localStorage.setItem(
              "illit_archive_members",
              data.members
            );

          }

        }

        if(
          data.calendar !==
          undefined
        ){

          if(data.calendar === null){

            localStorage.removeItem(
              "illit_archive_calendar"
            );

          }else{

            localStorage.setItem(
              "illit_archive_calendar",
              data.calendar
            );

          }

        }

        if(
          data.activities !==
          undefined
        ){

          if(data.activities === null){

            localStorage.removeItem(
              "illit_archive_activities"
            );

          }else{

            localStorage.setItem(
              "illit_archive_activities",
              data.activities
            );

          }

        }

        if(
          data.logo !==
          undefined
        ){

          if(data.logo === null){

            localStorage.removeItem(
              SETTINGS_LOGO_STORAGE
            );

          }else{

            localStorage.setItem(
              SETTINGS_LOGO_STORAGE,
              data.logo
            );

          }

        }

        closeModal();

        if(
          typeof renderProfileCards ===
          "function"
        ){

          renderProfileCards();

        }

        if(
          typeof renderCalendar ===
          "function"
        ){

          renderCalendar();

        }

        if(
          typeof renderActivities ===
          "function"
        ){

          renderActivities();

        }

        if(
          typeof renderStatus ===
          "function"
        ){

          renderStatus();

        }

        if(
          typeof updateHeaderLogo ===
          "function"
        ){

          updateHeaderLogo();

        }

        if(
          typeof createSettingsPage ===
          "function"
        ){

          createSettingsPage();

        }

        showToast(
          "Backup restored."
        );

      }catch(error){

        showToast(
          "Invalid backup file."
        );

      }

    };

  reader.readAsText(
    file
  );

}

<script>

/* =========================

const ARCHIVE_PAGES = [

  "home",
  "profile",
  "calendar",
  "activities",
  "status"

];

/* =========================
   NAVIGATION LABELS
========================= */

const NAVIGATION_ITEMS = [

  {
    id:"home",
    label:"Home",
    icon:"⌂"
  },

  {
    id:"profile",
    label:"Profile",
    icon:"♡"
  },

  {
    id:"calendar",
    label:"Calendar",
    icon:"▣"
  },

  {
    id:"activities",
    label:"Activities",
    icon:"▤"
  },

  {
    id:"status",
    label:"Status",
    icon:"⌁"
  }

];

/* =========================
   SET ACTIVE NAV
========================= */

function updateBottomNavigation(
  page
){

  const navButtons =
    document.querySelectorAll(
      ".bottom-nav button[data-page]"
    );

  navButtons.forEach(
    button => {

      const isActive =
        button.dataset.page ===
        page;

      button.classList.toggle(
        "active",
        isActive
      );

    }
  );

}

/* =========================
   SETTINGS BUTTON
========================= */

function goToSettings(){

  if(
    typeof showPage ===
    "function"
  ){

    showPage(
      "settings"
    );

  }

}

/* =========================
   FLOATING ACTION
========================= */

function handleFloatingAdd(){

  const activePage =
    document.querySelector(
      ".page.active"
    );

  if(!activePage){

    return;

  }

  const id =
    activePage.id;

  if(
    id ===
    "page-calendar"
  ){

    openCalendarAdd();

    return;

  }

  if(
    id ===
    "page-activities"
  ){

    openActivityAdd();

    return;

  }

  if(
    id ===
    "page-profile"
  ){

    showToast(
      "Use Edit on a member profile."
    );

    return;

  }

  if(
    id ===
    "page-status"
  ){

    showToast(
      "Add activities to update Status."
    );

    return;

  }

  showToast(
    "Choose a section to add content."
  );

}

/* =========================
   CREATE BOTTOM NAV
========================= */

function createBottomNavigation(){

  let nav =
    document.querySelector(
      ".bottom-nav"
    );

  if(!nav){

    nav =
      document.createElement(
        "nav"
      );

    nav.className =
      "bottom-nav";

    document.body.appendChild(
      nav
    );

  }

  nav.innerHTML =
    NAVIGATION_ITEMS.map(
      item => `

        <button

          type="button"

          data-page="${item.id}"

          onclick="
            showPage(
              '${item.id}'
            )
          "

        >

          <span class="bottom-nav-icon">
            ${item.icon}
          </span>

          <span class="bottom-nav-label">
            ${item.label}
          </span>

        </button>

      `
    ).join("");

}

/* =========================
   CREATE FLOATING BUTTON
========================= */

function createFloatingButton(){

  let button =
    document.querySelector(
      ".floating-add"
    );

  if(!button){

    button =
      document.createElement(
        "button"
      );

    button.className =
      "floating-add";

    button.type =
      "button";

    button.setAttribute(
      "aria-label",
      "Add"
    );

    button.onclick =
      handleFloatingAdd;

    document.body.appendChild(
      button
    );

  }

  button.textContent =
    "+";

}

/* =========================
   HEADER LOGO
========================= */

function updateHeaderLogo(){

  const logo =
    localStorage.getItem(
      SETTINGS_LOGO_STORAGE
    );

  const image =
    document.querySelector(
      ".app-header-logo img"
    );

  const fallback =
    document.querySelector(
      ".app-header-logo-fallback"
    );

  if(
    !image ||
    !fallback
  ){

    return;

  }

  if(logo){

    image.src =
      logo;

    image.style.display =
      "block";

    fallback.style.display =
      "none";

  }else{

    image.removeAttribute(
      "src"
    );

    image.style.display =
      "none";

    fallback.style.display =
      "block";

  }

}

/* =========================
   ENSURE HEADER
========================= */

function ensureAppHeader(){

  let header =
    document.querySelector(
      ".app-header"
    );

  if(!header){

    header =
      document.createElement(
        "header"
      );

    header.className =
      "app-header";

    document.body.insertBefore(
      header,
      document.body.firstChild
    );

  }

  header.innerHTML = `

    <div class="app-header-left">

      <div class="app-header-logo">

        <img
          alt="ILLIT Archive logo"
          style="display:none"
        >

        <span
          class="app-header-logo-fallback"
        >
          ✦

      </div>

        <div class="app-header-title">

    <button

      onclick="goToSettings()"
      aria-label="Settings"
    >
      ⚙
    </button>

  `;

  updateHeaderLogo();

}

/* =========================
   PAGE TITLE
========================= */

function updatePageHeading(
  page
){

  const headings = {

    home:[
      "Home",
      "this is our world"
    ],

    profile:[
      "Profile",
      "five members"
    ],

    calendar:[
      "Calendar",
      "group & individual activities"
    ],

    activities:[
      "Activities",
      "individual archive"
    ],

    status:[
      "Status",
      "activity evolution"
    ],

    settings:[
      "Settings",
      "archive management"
    ]

  };

  const values =
    headings[page];

  if(!values){
    return;
  }

  const currentPage =
    document.getElementById(
      "page-" + page
    );

  if(!currentPage){
    return;
  }

  const heading =
    currentPage.querySelector(
      ".page-heading"
    );

  if(!heading){
    return;
  }

  const h1 =
    heading.querySelector(
      "h1"
    );

  const p =
    heading.querySelector(
      "p"
    );

  if(h1){

    h1.textContent =
      values[0];

  }

  if(p){

    p.textContent =
      values[1];

  }

}

/* =========================
   FINAL SHOW PAGE
========================= */

const integratedShowPage =
  window.showPage;

window.showPage =
  function(page){

    if(
      page ===
      "settings"
    ){

      integratedShowPage(
        page
      );

      updateBottomNavigation(
        ""
      );

      return;

    }

    integratedShowPage(
      page
    );

    updateBottomNavigation(
      page
    );

    updatePageHeading(
      page
    );

    if(
      page ===
      "profile" &&
      typeof renderProfileCards ===
      "function"
    ){

      renderProfileCards();

    }

    if(
      page ===
      "calendar" &&
      typeof renderCalendar ===
      "function"
    ){

      renderCalendar();

    }

    if(
      page ===
      "activities" &&
      typeof renderActivities ===
      "function"
    ){

      renderActivities();

    }

    if(
      page ===
      "status" &&
      typeof renderStatus ===
      "function"
    ){

      renderStatus();

    }

  };

/* =========================
   INITIALIZE APP
========================= */

function initializeArchiveApp(){

  ensureAppHeader();

  createBottomNavigation();

  createFloatingButton();

  updateBottomNavigation(
    "home"
  );

  updatePageHeading(
    "home"
  );

  updateHeaderLogo();

  if(
    typeof createSettingsPage ===
    "function"
  ){

    createSettingsPage();

  }

}

document.addEventListener(
  "DOMContentLoaded",
  initializeArchiveApp
);
