    const locEl   = document.getElementById("nextEventLocation");
    if (!daysEl || !titleEl || !locEl) return;

      daysEl.textContent = "—";
      titleEl.textContent = "Sem eventos futuros";
      locEl.textContent = "";
      return;

    const days = Math.ceil((upcoming.d - now) / (1000 * 60 * 60 * 24));
    daysEl.textContent  = days === 1 ? "1 dia" : `${days} dias`;
    titleEl.textContent = upcoming.title;
    locEl.textContent   = `${upcoming.country} ${upcoming.location}`;

  function renderCalendarTabs() {
    const tabs = document.getElementById("calTabs");
    if (!tabs) return;

    const all = [
      { id: "ALL", label: "✨ ILLIT" },
      ...MEMBERS.map(m => ({ id: m.id, label: `${m.emoji} ${m.id} ${m.accent}` }))
    ];

    tabs.innerHTML = all.map(t => `
      <button class="tab ${currentFilter === t.id ? "active" : ""}"
              onclick="selectFilter('${t.id}')">
        ${t.label}
      </button>
    `).join("");
  }

  function selectFilter(id) {
    currentFilter = id;
    renderCalendar();
  }

  function renderCalendar() {
    renderCalendarTabs();

    const list = document.getElementById("calendarList");
    if (!list) return;

    let filtered = ACTIVITIES;
    if (currentFilter !== "ALL") {
      filtered = ACTIVITIES.filter(a => a.participants && a.participants.includes(currentFilter));
    }

    if (!filtered.length) {
      list.innerHTML = `
        <div class="empty">
          <div class="empty-icon">📅</div>
          Nenhuma atividade encontrada.
        </div>
      `;
      return;
    }

    const sorted = [...filtered].sort((a, b) => new Date(a.date) - new Date(b.date));

    list.innerHTML = sorted.map(a => {
      const d = new Date(a.date + "T00:00:00");
      const day = d.getDate();
      const month = d.toLocaleDateString("pt-BR", { month: "short" });

      return `
        <div class="event" onclick="openEventDetail('${a.id}')">
          <div class="event-date">
            <div class="event-date-day">${day}</div>
            <div class="event-date-month">${month}</div>
          </div>
          <div class="event-info">
            <h3>${a.title}</h3>
            <p>${a.country} ${a.location}</p>
            <span class="event-type">${a.type}</span>
          </div>
        </div>
      `;
    }).join("");
  }

  function renderStats() {
    const totalEl = document.getElementById("statsTotal");
    const listEl  = document.getElementById("statsList");
    const graphEl = document.getElementById("statsGraph");

    if (totalEl) totalEl.textContent = ACTIVITIES.length;

    /* Barras por membro */
    if (listEl) {
      const counts = MEMBERS.map(m => ({
        ...m,
        count: ACTIVITIES.filter(a => a.participants && a.participants.includes(m.id)).length
      }));

      const max = Math.max(...counts.map(c => c.count), 1);

      listEl.innerHTML = counts.map(m => {
        const pct = (m.count / max) * 100;
        return `
          <div class="card" style="margin-bottom: 10px;">
            <div style="display: flex; justify-content: space-between; align-items: center;">
              <strong>${m.emoji} ${m.id} ${m.accent}</strong>
              <span style="font-weight: 700;">${m.count}</span>
            </div>
            <div class="stat-bar">
              <div class="stat-fill" style="width: ${pct}%; background: ${m.color};"></div>
            </div>
          </div>
        `;
      }).join("");
    }

    /* Gráfico por mês */
    if (graphEl) {
      const monthCounts = {};

      ACTIVITIES.forEach(a => {
        const d = new Date(a.date + "T00:00:00");
        const key = d.toLocaleDateString("pt-BR", { month: "short" });
        monthCounts[key] = (monthCounts[key] || 0) + 1;
      });

      const entries = Object.entries(monthCounts);

      if (!entries.length) {
        graphEl.innerHTML = `<div style="width: 100%; text-align: center; color: var(--muted); font-size: 12px;">Sem dados</div>`;
      } else {
        const max = Math.max(...entries.map(e => e[1]), 1);

  function renderSearch() {
    const input = document.getElementById("searchInput");
    const results = document.getElementById("searchResults");
    if (!input || !results) return;

    /* Remove listener antigo clonando */
    const newInput = input.cloneNode(true);
    input.parentNode.replaceChild(newInput, input);

    newInput.addEventListener("input", (e) => {
      const q = e.target.value.trim().toLowerCase();

      if (!q) {
        results.innerHTML = `
          <div class="empty">
            <div class="empty-icon">🔎</div>
            Digite para buscar.
          </div>
        `;
        return;
      }

      const items = [];

      /* Atividades */
      ACTIVITIES.forEach(a => {
        items.push({
          type: "activity",
          id: a.id,
          title: a.title,
          meta: `${a.date} · ${a.type}`,
          emoji: "📅"
        });
      });

      /* Membros */
      MEMBERS.forEach(m => {
        items.push({
          type: "member",
          id: m.id,
          title: `${m.id} ${m.accent}`,
          meta: `Membro · ${m.birthday}`,
          emoji: m.emoji
        });
      });

      /* Magazines */
      MAGAZINES.forEach(m => {
        items.push({
          type: "magazine",
          id: m.id,
          title: m.name,
          meta: `Magazine · ${m.member}`,
          emoji: "📰"
        });
      });

      /* Ambassadors */
      AMBASSADORS.forEach(a => {
        items.push({
          type: "ambassador",
          id: a.id,
          title: a.brand,
          meta: `Ambassador · ${a.member}`,
          emoji: "🤝"
        });
      });

      const matches = items.filter(i =>
        (i.title + " " + i.meta).toLowerCase().includes(q)
      );

      if (!matches.length) {
        results.innerHTML = `
          <div class="empty">
            <div class="empty-icon">🫧</div>
            Nada encontrado.
          </div>
        `;
        return;
      }

      results.innerHTML = matches.map(m => `
        <div class="search-result" onclick="gotoSearchResult('${m.type}', '${m.id}')">
          <div class="search-result-icon">${m.emoji}</div>
          <div class="search-result-info">
            <div class="search-result-title">${m.title}</div>
            <div class="search-result-meta">${m.meta}</div>
          </div>
        </div>
      `).join("");
    });

    newInput.value = "";
    newInput.dispatchEvent(new Event("input"));
  }

  function gotoSearchResult(type, id) {
    if (type === "activity") {
      showSection("calendarSection");
      setTimeout(() => openEventDetail(id), 300);
    } else if (type === "member") {
      showSection("profileSection");
    } else if (type === "magazine") {
      showSection("magazinesSection");
    } else if (type === "ambassador") {
      showSection("ambassadorsSection");
    }
  }

  function toggleFavorite(type, id, title, meta, image) {
    const idx = FAVORITES.findIndex(f => f.type === type && String(f.id) === String(id));

    if (idx >= 0) {
      FAVORITES.splice(idx, 1);
      showToast("Removido dos favoritos");
    } else {
      FAVORITES.push({ type, id, title, meta, image });
      showToast("⭐ Adicionado aos favoritos");
    }

    saveAll();
    renderFavorites();

    /* Atualiza modal se estiver aberto */
    const modalActive = document.getElementById("modal")?.classList.contains("active");
    if (modalActive && type === "activity") {
      setTimeout(() => openEventDetail(id), 100);
    }
  }

  function renderFavorites() {
    const list = document.getElementById("favoritesList");
    if (!list) return;

    if (!FAVORITES.length) {
      list.innerHTML = `
        <div class="empty">
          <div class="empty-icon">⭐</div>
          Você ainda não salvou nada.<br>
          <small>Toque em ☆ nas atividades para salvar.</small>
        </div>
      `;
      return;
    }

    list.innerHTML = FAVORITES.map(f => {
      const emoji = f.type === "activity" ? "📅"
                  : f.type === "member"   ? "👤"
                  : f.type === "magazine" ? "📰"
                  : "🤝";

      return `
        <div class="fav-card">
          <div class="search-result-icon">${emoji}</div>
          <div class="fav-info">
            <div class="fav-title">${f.title}</div>
            <div class="fav-meta">${f.meta || f.type}</div>
          </div>
          <button class="admin-icon danger" onclick="toggleFavorite('${f.type}', '${f.id}')">🗑️</button>
        </div>
      `;
    }).join("");
  }
