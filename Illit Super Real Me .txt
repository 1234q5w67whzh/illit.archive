function showSection(sectionId, clickedButton = null) {

  document
    .querySelectorAll(".page-section")
    .forEach(section => {
      section.classList.remove("active");
    });

  const target = document.getElementById(sectionId);

  if (target) {
    target.classList.add("active");
  }

  document
    .querySelectorAll(".nav-item")
    .forEach(button => {
      button.classList.remove("active");
    });

  const navButton =
    clickedButton ||
    document.querySelector(
      `.nav-item[data-section="${sectionId}"]`
    );

  if (navButton) {
    navButton.classList.add("active");
  }

  window.scrollTo({
    top: 0,
    behavior: "smooth"
  });
}

}

/* ==========================================
   COUNTDOWN
========================================== */

function updateCountdown() {

  const now = new Date();

  const upcoming =
    ILLIT_REAL_ACTIVITIES
      .map(event => ({
        ...event,
        start: new Date(event.date + "T00:00:00")
      }))
      .filter(event => event.start > now)
      .sort((a,b) => a.start - b.start)[0];

  const number =
    document.getElementById("countdownNumber");

  const title =
    document.getElementById("countdownEvent");

  const location =
    document.getElementById("countdownLocation");

  if (!upcoming) {

    number.textContent = "—";
    title.textContent = "No upcoming events";
    location.textContent = "";

    return;
  }

  const difference =
    upcoming.start - now;

  const days =
    Math.ceil(
      difference / (1000 * 60 * 60 * 24)
    );

  number.textContent =
    days === 1
      ? "1 day"
      : `${days} days`;

  title.textContent =
    upcoming.title;

  location.textContent =
    `${upcoming.country} ${upcoming.location}`;
}

<script>

/* ==========================================
   CALENDAR STATE
========================================== */

let selectedCalendarMember = "ALL";

/* ==========================================
   DATE HELPERS
========================================== */

function getMonthName(dateString) {

  const date =
    new Date(dateString + "T00:00:00");

  return date.toLocaleDateString(
    "en-US",
    {
      month: "long",
      year: "numeric"
    }
  );
}

function getShortMonth(dateString) {

  const date =
    new Date(dateString + "T00:00:00");

  return date.toLocaleDateString(
    "en-US",
    {
      month: "short"
    }
  );
}

function getDay(dateString) {

  const date =
    new Date(dateString + "T00:00:00");

  return date.getDate();
}

/* ==========================================
   FILTER EVENTS
========================================== */

function getFilteredCalendarEvents() {

  if (selectedCalendarMember === "ALL") {
    return [...ILLIT_REAL_ACTIVITIES];
  }

  return ILLIT_REAL_ACTIVITIES.filter(event => {

    return (
      event.participants &&
      event.participants.includes(
        selectedCalendarMember
      )
    );

  });

}

/* ==========================================
   SELECT MEMBER TAB
========================================== */

function selectCalendarMember(
  memberId,
  button
) {

  selectedCalendarMember = memberId;

  document
    .querySelectorAll(".calendar-tab")
    .forEach(tab => {
      tab.classList.remove("active");
    });

  if (button) {
    button.classList.add("active");
  }

  renderCalendar();
}

/* ==========================================
   RENDER CALENDAR
========================================== */

function renderCalendar() {

  const container =
    document.getElementById(
      "calendarEventsContainer"
    );

  if (!container) return;

  const events =
    getFilteredCalendarEvents();

  if (!events.length) {

    container.innerHTML = `
      <div class="event-empty">
        No activities found for this category.
      </div>
    `;

    return;
  }

  const sortedEvents =
    [...events].sort(
      (a,b) =>
        new Date(a.date) -
        new Date(b.date)
    );

  const grouped = {};

  sortedEvents.forEach(event => {

    const month =
      getMonthName(event.date);

    if (!grouped[month]) {
      grouped[month] = [];
    }

    grouped[month].push(event);

  });

  container.innerHTML =
    Object.entries(grouped)
      .map(([month, monthEvents]) => {

        return `

          <div class="calendar-month">

            <div class="calendar-month-title">

              <h2>${month}</h2>

              <span>
                ${monthEvents.length}
                ${monthEvents.length === 1
                  ? "activity"
                  : "activities"}
              </span>

            </div>

            ${monthEvents
              .map(event => renderEventCard(event))
              .join("")}

          </div>

        `;

      })
      .join("");
}

/* ==========================================
   EVENT CARD
========================================== */

function renderEventCard(event) {

  const dateText =
    event.date === event.endDate
      ? getDay(event.date)
      : `${getDay(event.date)}–${getDay(event.endDate)}`;

  return `

    <article
      class="event-card"
      onclick="openEventDetail(${event.id})"
    >

      <div class="event-date">

        <div class="event-date-day">
          ${dateText}
        </div>

        <div class="event-date-month">
          ${getShortMonth(event.date)}
        </div>

      </div>

      <div class="event-info">

        <h3>
          ${event.title}
        </h3>

        <p>
          ${event.country}
          ${event.location}
        </p>

        <span class="event-type">
          ${event.type}
        </span>

      </div>

      <div class="event-arrow">
        ›
      </div>

    </article>

  `;
}

}

/* ==========================================
   GET FILTERED STAT EVENTS
========================================== */

function getStatisticsEvents() {

  const year =
    document.getElementById("statsYear")?.value ||
    "ALL";

  const month =
    document.getElementById("statsMonth")?.value ||
    "ALL";

  const type =
    document.getElementById("statsType")?.value ||
    "ALL";

  return ILLIT_REAL_ACTIVITIES.filter(event => {

    const date =
      new Date(event.date + "T00:00:00");

    const eventYear =
      String(date.getFullYear());

    const eventMonth =
      String(date.getMonth() + 1)
        .padStart(2, "0");

    if (
      year !== "ALL" &&
      eventYear !== year
    ) {
      return false;
    }

    if (
      month !== "ALL" &&
      eventMonth !== month
    ) {
      return false;
    }

    if (
      type !== "ALL" &&
      event.type !== type
    ) {
      return false;
    }

    return true;

  });

}

/* ==========================================
   MEMBER COUNTS
========================================== */

function calculateMemberCounts(events) {

  const counts = {
    YUNAH: 0,
    MINJU: 0,
    MOKA: 0,
    WONHEE: 0,
    IROHA: 0
  };

  events.forEach(event => {

    if (!event.participants) return;

    event.participants.forEach(participant => {

      if (
        Object.prototype.hasOwnProperty.call(
          counts,
          participant
        )
      ) {
        counts[participant]++;
      }

    });

  });

  return counts;

}

/* ==========================================
   RENDER MEMBER STATS
========================================== */

function renderMemberStatistics(events) {

  const container =
    document.getElementById(
      "memberStatsGrid"
    );

  if (!container) return;

  const counts =
    calculateMemberCounts(events);

  const max =
    Math.max(
      ...Object.values(counts),
      1
    );

  const totalMemberActivities =
    Object.values(counts)
      .reduce(
        (sum, value) => sum + value,
        0
      );

  container.innerHTML =
    Object.entries(counts)
      .map(([member, count]) => {

        const data =
          STAT_MEMBER_DATA[member];

        const percentage =
          totalMemberActivities
            ? (count / totalMemberActivities) * 100
            : 0;

        const width =
          (count / max) * 100;

        return `

          <article
            class="member-stat-card"
            style="
              border-top:3px solid ${data.color};
            "
          >

            <div class="member-stat-top">

              <div
                class="member-stat-icon ${data.background}"
              >
                ${data.emoji}
              </div>

              <div>
                <div class="member-stat-name">
                  ${member} ${data.accent}
                </div>
              </div>

            </div>

            <div class="member-stat-count">
              ${count}
            </div>

            <div class="member-stat-label">
              activities
            </div>

            <div class="stat-progress">

              <div
                class="stat-progress-fill"
                style="
                  width:${width}%;
                  background:${data.color};
                "
              ></div>

            </div>

            <div class="stat-percentage">
              ${percentage.toFixed(1)}% of member activities
            </div>

          </article>

        `;

      })
      .join("");

}

/* ==========================================
   FIND MOST / LEAST
========================================== */

function renderStatisticsAlerts(events) {

  const container =
    document.getElementById(
      "statsAlerts"
    );

  if (!container) return;

  const counts =
    calculateMemberCounts(events);

  const values =
    Object.values(counts);

  if (!values.some(value => value > 0)) {

    container.innerHTML = `
      <div class="stats-alert-title">
        INSIGHTS
      </div>

      <div class="stats-alert-main">
        No member activity data available.
      </div>
    `;

    return;
  }

  const max =
    Math.max(...values);

  const min =
    Math.min(
      ...values
        .filter(value => value >= 0)
    );

  const most =
    Object.keys(counts)
      .filter(
        member => counts[member] === max
      );

  const least =
    Object.keys(counts)
      .filter(
        member => counts[member] === min
      );

  const mostText =
    most
      .map(
        member =>
          `${STAT_MEMBER_DATA[member].accent} ${member}`
      )
      .join(", ");

  const leastText =
    least
      .map(
        member =>
          `${STAT_MEMBER_DATA[member].accent} ${member}`
      )
      .join(", ");

  container.innerHTML = `

    <div class="stats-alert-title">
      INSIGHTS
    </div>

    <div class="stats-alert-main">
      🏆 Most activities: ${mostText}
    </div>

    <div class="stats-alert-main">
      🚨 Least activities: ${leastText}
    </div>

  `;

}

/* ==========================================
   ACTIVITY TYPE STATS
========================================== */

function renderActivityTypeStats(events) {

  const container =
    document.getElementById(
      "activityTypeStats"
    );

  if (!container) return;

  const counts = {};

  events.forEach(event => {

    const type =
      event.type || "Other";

    counts[type] =
      (counts[type] || 0) + 1;

  });

  const entries =
    Object.entries(counts);

  if (!entries.length) {

    container.innerHTML = `
      <div class="stats-empty">
        No activity types available.
      </div>
    `;

    return;
  }

  const max =
    Math.max(
      ...entries.map(
        item => item[1]
      ),
      1
    );

  container.innerHTML =
    entries
      .sort((a,b) => b[1] - a[1])
      .map(([type,count]) => {

        const width =
          (count / max) * 100;

        return `

          <div class="type-stat-row">

            <div class="type-stat-name">
              ${type}
            </div>

            <div class="type-stat-bar">

              <div
                class="type-stat-fill"
                style="width:${width}%"
              ></div>

            </div>

            <div class="type-stat-number">
              ${count}
            </div>

          </div>

        `;

      })
      .join("");

}

/* ==========================================
   TIMELINE GRAPH
========================================== */

function renderStatisticsGraph(events) {

  const container =
    document.getElementById(
      "statsGraph"
    );

  if (!container) return;

  if (!events.length) {

    container.innerHTML = `
      <div class="stats-empty" style="width:100%;">
        No timeline data available.
      </div>
    `;

    return;
  }

  const monthCounts = {};

  events.forEach(event => {

    const date =
      new Date(
        event.date + "T00:00:00"
      );

    const key =
      date.toLocaleDateString(
        "en-US",
        {
          month: "short"
        }
      );

    monthCounts[key] =
      (monthCounts[key] || 0) + 1;

  });

  const entries =
    Object.entries(monthCounts);

  const max =
    Math.max(
      ...entries.map(
        item => item[1]
      ),
      1
    );

  container.innerHTML =
    entries
      .map(([month,count]) => {

        const height =
          Math.max(
            7,
            (count / max) * 125
          );

        return `

          <div class="graph-column">

            <div class="graph-number">
              ${count}
            </div>

            <div
              class="graph-bar"
              style="
                height:${height}px;
                background:var(--text);
              "
            ></div>

            <div class="graph-label">
              ${month}
            </div>

          </div>

        `;

      })
      .join("");

}

  }

  renderMemberStatistics(events);

  renderStatisticsAlerts(events);

  renderActivityTypeStats(events);

  renderStatisticsGraph(events);

}

  window.showSection;

window.showSection =
function(sectionId, clickedButton = null) {

  if (sectionId === "statsSection") {

    document
      .querySelectorAll(".page-section")
      .forEach(section => {
        section.classList.remove("active");
      });

    document
      .getElementById("realStatsScreen")
      .classList.add("active");

    document
      .querySelectorAll(".nav-item")
      .forEach(button => {
        button.classList.remove("active");
      });

    const statsButton =
      document.querySelector(
        '.nav-item[data-section="statsSection"]'
      );

    if (statsButton) {
      statsButton.classList.add("active");
    }

    renderStatistics();

    window.scrollTo({
      top: 0,
      behavior: "smooth"
    });

    return;
  }

  statisticsNavigationBase(
    sectionId,
    clickedButton
  );

};

  );

    "illit_admin_members",

  );

/* =========================
   SYNC SITE DATA
========================= */

function syncSiteData(){

  if(window.ILLIT_ADMIN_ACTIVITIES){

    try{
      ILLIT_REAL_ACTIVITIES.length = 0;

      window.ILLIT_ADMIN_ACTIVITIES.forEach(activity => {
        ILLIT_REAL_ACTIVITIES.push(activity);
      });
    }catch(error){}
  }

  if(window.ILLIT_ADMIN_MEMBERS){

    try{
      ILLIT_MEMBERS.length = 0;

      window.ILLIT_ADMIN_MEMBERS.forEach(member => {
        ILLIT_MEMBERS.push(member);
      });
    }catch(error){}
  }

  if(window.ILLIT_ADMIN_AMBASSADORS){

    try{
      ILLIT_AMBASSADORS.length = 0;

      window.ILLIT_ADMIN_AMBASSADORS.forEach(item => {
        ILLIT_AMBASSADORS.push(item);
      });
    }catch(error){}
  }

  if(window.ILLIT_ADMIN_MAGAZINES){

    try{
      ILLIT_MAGAZINES.length = 0;

      window.ILLIT_ADMIN_MAGAZINES.forEach(item => {
        ILLIT_MAGAZINES.push(item);
      });
    }catch(error){}
  }

  try{
    renderMagazineTabs();
    renderMagazines();
  }catch(error){}

  try{
    renderAmbassadorTabs();
    renderAmbassadors();
  }catch(error){}

  try{
    renderCalendar();
  }catch(error){}

  try{
    renderStats();
  }catch(error){}
}

/* =========================
   LOGIN ACTION
========================= */

function adminLogin(){

  const input =
    document.getElementById("adminPasswordInput");

  const error =
    document.getElementById("adminLoginError");

  if(!input) return;

  if(input.value === ADMIN_PASSWORD){

    adminLoggedIn = true;

    renderAdminPanel();

  }else{

    if(error){
      error.textContent =
        "Incorrect password.";
    }

    input.value = "";
    input.focus();

}

    : {};

    prompt(

    );

  if(title === null) return;

    prompt(
      "Date:",
      activity.date || ""
    );

  if(date === null) return;

    prompt(
      "Time:",
      activity.time || ""
    );

  if(time === null) return;

  const location =
    prompt(
      "Location:",
      activity.location || ""
    );

  if(location === null) return;

  const country =
    prompt(
      "Country:",
      activity.country || ""
    );

  if(country === null) return;

  const type =
    prompt(
      "Activity type:",
      activity.type || "Event"
    );

  if(type === null) return;

  const participants =
    prompt(
      "Participants:",
      activity.participants || "ILLIT"
    );

  if(participants === null) return;

  if(status === null) return;

      activity.image || ""
    );

  if(image === null) return;

  const link =
    prompt(

      activity.link || ""
    );

window.showSection = function(section){

  if(section === "adminSection"){

    document
      .querySelectorAll(".screen")
      .forEach(screen =>
        screen.classList.remove("active")
      );

    const target =
      document.getElementById("realAdminScreen");

    if(target){

      target.classList.add("active");

      if(adminLoggedIn){
        renderAdminPanel();

        renderAdminLogin();
      }

    }

    window.scrollTo({
      top:0,
      behavior:"smooth"
    });

    return;
  }

  return adminNavigationBase(section);

/* =========================
   IMAGE FILE READER
========================= */

function readAdminImage(file, callback){

  if(!file){
    callback("");
    return;
  }

  const reader = new FileReader();

  reader.onload = function(event){

    callback(event.target.result);

  };

  reader.readAsDataURL(file);
}

/* =========================
   IMAGE PREVIEW
========================= */

function previewAdminImage(
  inputId,
  previewId
){

  const input =
    document.getElementById(inputId);

  const preview =
    document.getElementById(previewId);

  if(!input || !preview) return;

  input.addEventListener(
    "change",
    function(){

      const file =
        this.files && this.files[0];

      if(!file) return;

      const reader =
        new FileReader();

      reader.onload =
        function(event){

          preview.innerHTML = `
            <img
              src="${event.target.result}"
              alt="Preview">
          `;

        };

      reader.readAsDataURL(file);

    }
  );
}

      if(
        participants.includes("IROHA") &&
        !participants.includes("YUNAH") &&
        !participants.includes("MINJU") &&
        !participants.includes("MOKA") &&
        !participants.includes("WONHEE")
      ){
        member = "IROHA";
      }

      photos.push({

        image:activity.image,

        title:
          activity.title ||
          "ILLIT Activity",

        member

      });

    });

  /* Magazine covers and galleries */

  (window.ILLIT_ADMIN_MAGAZINES || [])
    .forEach(magazine => {

      const member =
        magazine.member || "ILLIT";

      if(magazine.cover){

        photos.push({

          image:magazine.cover,

          title:
            magazine.name ||
            "Magazine",

          member

        });

      }

      if(Array.isArray(magazine.gallery)){

        magazine.gallery.forEach(image => {

          if(!image) return;

          photos.push({

            image,

            title:
              magazine.name ||
              "Photoshoot",

            member

          });

        });

      }

    });

  /* Ambassador campaign images */

  (window.ILLIT_ADMIN_AMBASSADORS || [])
    .forEach(item => {

      if(item.image){

        photos.push({

          image:item.image,

          title:
            item.brand ||
            "Ambassador Campaign",

          member:
            item.member ||
            "ILLIT"

        });

      }

    });

  /* Member profile photos */

  (window.ILLIT_ADMIN_MEMBERS || [])
    .forEach(member => {

      if(member.photo){

        photos.push({

          image:member.photo,

          title:
            member.name ||
            member.id,

          member:
            member.id ||
            "ILLIT"

        });

      }

    });

  return photos;
}

function performArchiveSearch(query){

  const results =
    document.getElementById(
      "searchResults"
    );

  if(!results) return;

  const cleanQuery =
    String(query || "")
      .trim()
      .toLowerCase();

  if(!cleanQuery){

    results.innerHTML = `

      <div class="search-empty">

        <div class="search-empty-icon">
          🔎
        </div>

        <div>
          Start typing to search
          the archive.
        </div>

      </div>

    `;

    return;
  }

  const items =
    getArchiveSearchItems();

  const matches =
    items.filter(item => {

      const text = [

        item.title,
        item.meta,
        item.type

      ]
      .join(" ")
      .toLowerCase();

      return text.includes(
        cleanQuery
      );

    });

  if(!matches.length){

    results.innerHTML = `

      <div class="search-empty">

        <div class="search-empty-icon">
          🫧
        </div>

        <div>
          No results found.
        </div>

      </div>

    `;

    return;
  }

  results.innerHTML =
    matches.map((item,index) => `

      <div
        class="search-result"
        onclick="
          executeSearchResult(
            ${index},
            ${JSON.stringify(
              matches.map(item => ({
                type:item.type,
                title:item.title,
                meta:item.meta,
                image:item.image
              }))
            )}
          )
        ">

        <div class="search-result-icon">

          ${
            item.image
            ?
            `<img
              src="${item.image}"
              alt="">`
            :
            (
              item.type === "Activity"
              ? "📅"
              : item.type === "Member"
              ? "👤"
              : item.type === "Magazine"
              ? "📰"
              : "🤝"
            )
          }

        </div>

        <div class="search-result-main">

          <div class="search-result-title">
            ${item.title}
          </div>

          <div class="search-result-meta">
            ${item.meta || "No additional information"}
          </div>

        </div>

        <div class="search-type">
          ${item.type}
        </div>

      </div>

    `).join("");
}

/* =========================
   SEARCH RESULT ACTION
========================= */

function executeSearchResult(
  index,
  serializedMatches
){

  const items =
    getArchiveSearchItems();

  const data =
    serializedMatches[index];

  if(!data) return;

  const matching =
    items.find(item =>
      item.type === data.type &&
      item.title === data.title &&
      item.meta === data.meta
    );

  if(
    matching &&
    typeof matching.action === "function"
  ){

    matching.action();

  }

}

  "illit_archive_favorites";

    )
  );

}

/* =========================
   CREATE FAVORITE ID
========================= */

function createFavoriteId(
  type,
  id
){

  return String(type) +
    "::" +
    String(id);

}

/* =========================
   ADD FAVORITE
========================= */

function addFavorite(
  type,
  id,
  title,
  meta,
  image
){

  const favoriteId =
    createFavoriteId(
      type,
      id
    );

  const exists =
    ILLIT_FAVORITES.some(
      item =>
        item.favoriteId ===
        favoriteId
    );

  if(exists) return;

  ILLIT_FAVORITES.push({

    favoriteId,

    type,

    id,

    title:
      title || "Untitled",

    meta:
      meta || "",

    image:
      image || ""

  });

  saveFavorites();

  renderFavorites();

}

/* =========================
   REMOVE FAVORITE
========================= */

function removeFavorite(
  favoriteId
){

  ILLIT_FAVORITES =
    ILLIT_FAVORITES.filter(
      item =>
        item.favoriteId !==
        favoriteId
    );

  saveFavorites();

  renderFavorites();

}

/* =========================
   IS FAVORITE
========================= */

function isFavorite(
  type,
  id
){

  const favoriteId =
    createFavoriteId(
      type,
      id
    );

  return ILLIT_FAVORITES.some(
    item =>
      item.favoriteId ===
      favoriteId
  );

}

/* =========================
   TOGGLE FAVORITE
========================= */

function toggleFavorite(
  type,
  id,
  title,
  meta,
  image
){

  const favoriteId =
    createFavoriteId(
      type,
      id
    );

  if(
    isFavorite(
      type,
      id
    )
  ){

    removeFavorite(
      favoriteId
    );

  }else{

    addFavorite(
      type,
      id,
      title,
      meta,
      image
    );

  }

}

/* =========================
   PARSE ACTIVITY DATE
========================= */

function getHomeActivityDate(
  activity
){

  if(!activity) return null;

  const raw =
    activity.date ||
    activity.startDate ||
    "";

  const match =
    String(raw).match(
      /(\d{1,2})[\/\-](\d{1,2})/
    );

  if(match){

    const month =
      Number(match[1]);

    const day =
      Number(match[2]);

    const now =
      new Date();

    let year =
      now.getFullYear();

    return new Date(
      year,
      month - 1,
      day
    );

  }

  const parsed =
    new Date(raw);

  return isNaN(parsed.getTime())
    ? null
    : parsed;

}

/* =========================
   COUNTDOWN
========================= */

let homeCountdownTimer = null;

function updateHomeCountdown(
  targetDate
){

  if(homeCountdownTimer){

    clearInterval(
      homeCountdownTimer
    );

  }

  function update(){

    const now =
      new Date();

    const distance =
      targetDate - now;

    if(distance <= 0){

      renderHomeNextEvent();

      return;

    }

    const days =
      Math.floor(
        distance /
        (1000 * 60 * 60 * 24)
      );

    const hours =
      Math.floor(
        (distance /
        (1000 * 60 * 60)) % 24
      );

    const minutes =
      Math.floor(
        (distance /
        (1000 * 60)) % 60
      );

    const seconds =
      Math.floor(
        (distance / 1000) % 60
      );

    const d =
      document.getElementById(
        "homeDays"
      );

    const h =
      document.getElementById(
        "homeHours"
      );

    const m =
      document.getElementById(
        "homeMinutes"
      );

    const s =
      document.getElementById(
        "homeSeconds"
      );

    if(d) d.textContent =
      String(days);

    if(h) h.textContent =
      String(hours).padStart(
        2,"0"
      );

    if(m) m.textContent =
      String(minutes).padStart(
        2,"0"
      );

    if(s) s.textContent =
      String(seconds).padStart(
        2,"0"
      );

  }

  update();

  homeCountdownTimer =
    setInterval(
      update,
      1000
    );

}

      .toUpperCase();

/* =========================
   PARTICIPANT CHECK
========================= */

function activityIncludesMember(
  activity,
  memberId
){

  const participants =
    activity &&
    activity.participants;

  if(Array.isArray(participants)){

    return participants
      .map(String)
      .some(
        name =>
          name
            .toUpperCase()
            .includes(memberId)
      );

  }

  return String(
    participants || ""
  )
  .toUpperCase()
  .includes(memberId);

}

/* =========================
   DYNAMIC STAT CALCULATION
========================= */

function calculateDynamicStats(
  activities
){

  const counts = {};

  DYNAMIC_STATS_MEMBERS
    .forEach(member => {

      counts[member.id] =
        activities.filter(
          activity =>
            activityIncludesMember(
              activity,
              member.id
            )
        ).length;

    });

  const total =
    Object.values(counts)
      .reduce(
        (sum,value) =>
          sum + value,
        0
      );

  return {
    counts,
    total
  };

}

/* =========================
   PERCENTAGES
========================= */

function calculateMemberPercentages(
  counts
){

  const total =
    Object.values(counts)
      .reduce(
        (sum,value) =>
          sum + value,
        0
      );

  const percentages = {};

  DYNAMIC_STATS_MEMBERS
    .forEach(member => {

      percentages[member.id] =
        total > 0
        ?
        (counts[member.id] /
          total) * 100
        :
        0;

    });

  return percentages;

}

/* =========================
   RANK MEMBERS
========================= */

function getDynamicRanking(
  counts
){

  return [
    ...DYNAMIC_STATS_MEMBERS
  ]
  .sort(
    (a,b) =>
      counts[b.id] -
      counts[a.id]
  );

}

/* =========================
   RENDER DYNAMIC STATS
========================= */

function renderDynamicMemberStats(
  activities
){

  const container =
    document.getElementById(
      "dynamicMemberStats"
    );

  const alerts =
    document.getElementById(
      "dynamicStatAlerts"
    );

  if(!container) return;

  const result =
    calculateDynamicStats(
      activities
    );

  const percentages =
    calculateMemberPercentages(
      result.counts
    );

  const ranking =
    getDynamicRanking(
      result.counts
    );

  const highest =
    ranking[0];

  const lowest =
    ranking[
      ranking.length - 1
    ];

  if(alerts){

    alerts.innerHTML = `

      <div class="dynamic-alert">

        <div class="dynamic-alert-label">
          ✨ Most activities
        </div>

        <div
          class="dynamic-alert-name">

          ${
            highest.emoji
          }

          ${highest.id}

        </div>

      </div>

      <div class="dynamic-alert">

        <div class="dynamic-alert-label">
          🚨 Least activities
        </div>

        <div
          class="dynamic-alert-name">

          ${
            lowest.emoji
          }

          ${lowest.id}

        </div>

      </div>

    `;

  }

  container.innerHTML =
    ranking.map(
      (member,index) => {

        const count =
          result.counts[
            member.id
          ];

        const percentage =
          percentages[
            member.id
          ];

        const isTop =
          index === 0 &&
          count > 0;

        const isLow =
          index ===
          ranking.length - 1 &&
          count ===
          Math.min(
            ...Object.values(
              result.counts
            )
          );

        return `

          <div
            class="
              dynamic-stat-card
              ${
                isTop
                ? "top-member"
                : ""
              }
              ${
                isLow && !isTop
                ? "low-member"
                : ""
              }
            ">

            <div
              class="dynamic-stat-header">

              <div
                class="dynamic-stat-name">

                <span
                  class="
                    member-color-dot
                    ${getMemberClass(
                      member.id
                    )}
                  ">
                </span>

                ${
                  member.emoji
                }

                ${member.id}

              </div>

              <div
                class="dynamic-stat-percent">

                ${
                  percentage
                    .toFixed(1)
                }%

              </div>

            </div>

            <div
              class="dynamic-stat-track">

              <div
                class="dynamic-stat-fill"
                style="
                  width:
                    ${percentage}%;
                  background:
                    ${member.color};
                ">
              </div>

            </div>

            <div
              class="dynamic-stat-header"
              style="
                margin-top:7px;
                font-size:11px;
                color:var(--muted,#777);
              ">

              <span>
                ${count} activities
              </span>

              ${
                isTop
                ?
                `<span>✨ Leading</span>`
                :
                isLow
                ?
                `<span>Faded</span>`
                :
                `<span></span>`
              }

            </div>

          </div>

        `;

      }
    ).join("");

}

/* =========================
   ADD FAVORITES TO ACTIVITY
   CARDS
========================= */

function enhanceActivityCards(){

  const activities =
    window.ILLIT_ADMIN_ACTIVITIES ||
    window.ILLIT_REAL_ACTIVITIES ||
    [];

  document
    .querySelectorAll(
      ".activity-card"
    )
    .forEach((card,index) => {

      if(
        card.querySelector(
          "[data-archive-favorite]"
        )
      ) return;

      const activity =
        activities[index];

      if(!activity) return;

      const id =
        activity.id ||
        index;

      const button =
        document.createElement(
          "button"
        );

      button.className =
        "ios-action-button";

      button.dataset
        .archiveFavorite =
        "true";

      button.dataset
        .favoriteType =
        "Activity";

      button.dataset
        .favoriteId =
        String(id);

      const active =
        isArchiveFavorite(
          "Activity",
          id
        );

      button.classList.toggle(
        "favorite-active",
        active
      );

      button.innerHTML =
        `<span>${
          active ? "⭐" : "☆"
        }</span> ${
          active ? "Saved" : "Save"
        }`;

      button.onclick =
        function(event){

          event.stopPropagation();

          toggleArchiveFavorite(
            "Activity",
            id,
            activity.title ||
              "Activity",
            [
              activity.date,
              activity.location
            ]
            .filter(Boolean)
            .join(" · "),
            activity.image ||
              ""
          );

          refreshFavoriteButtons();

          showArchiveToast(
            isArchiveFavorite(
              "Activity",
              id
            )
            ? "Saved to Favorites"
            : "Removed from Favorites"
          );

        };

      const actionTarget =
        card.querySelector(
          ".event-action-bar"
        ) ||
        card;

      actionTarget.appendChild(
        button
      );

    });

}

/* =========================
   THEME
========================= */

function toggleArchiveTheme(){

  const current =
    document.documentElement
      .getAttribute("data-theme");

  const dark =
    current !== "dark";

  document.documentElement
    .setAttribute(
      "data-theme",
      dark
      ? "dark"
      : "light"
    );

  localStorage.setItem(
    "illit_archive_theme",
    dark
    ? "dark"
    : "light"
  );

  updateSettingsUI();

}

/* =========================
   ANIMATIONS
========================= */

function toggleArchiveAnimations(){

  ILLIT_SETTINGS.animations =
    !ILLIT_SETTINGS.animations;

  saveArchiveSettings();

  applyArchiveSettings();

  updateSettingsUI();

}

/* =========================
   COMPACT MODE
========================= */

function toggleArchiveCompact(){

  ILLIT_SETTINGS.compactMode =
    !ILLIT_SETTINGS.compactMode;

  saveArchiveSettings();

  applyArchiveSettings();

  updateSettingsUI();

}

/* =========================
   CLEAR FAVORITES
========================= */

function clearArchiveFavorites(){

  if(
    !ILLIT_FAVORITES.length
  ){

    showArchiveToast(
      "No favorites to clear"
    );

    return;

  }

  const confirmed =
    confirm(
      "Clear all saved favorites?"
    );

  if(!confirmed) return;

  ILLIT_FAVORITES = [];

  saveFavorites();

  renderFavorites();

  showArchiveToast(
    "Favorites cleared"
  );

}

/* =========================
   SETTINGS UI
========================= */

function updateSettingsUI(){

  const themeButton =
    document.getElementById(
      "settingsThemeButton"
    );

  const animationButton =
    document.getElementById(
      "settingsAnimationButton"
    );

  const compactButton =
    document.getElementById(
      "settingsCompactButton"
    );

  const theme =
    document.documentElement
      .getAttribute(
        "data-theme"
      );

  if(themeButton){

    themeButton.classList.toggle(
      "active",
      theme === "dark"
    );

  }

  if(animationButton){

    animationButton.classList.toggle(
      "active",
      ILLIT_SETTINGS.animations
    );

  }

  if(compactButton){

    compactButton.classList.toggle(
      "active",
      ILLIT_SETTINGS.compactMode
    );

  }

}

========================= */

    );

  if(savedTheme){

    document.documentElement

        savedTheme
      );

  }

  applyArchiveSettings();

})();

function renderSmartSearch(){

  const input =
    document.getElementById(
      "smartSearchInput"
    );

  const results =
    document.getElementById(
      "smartSearchResults"
    );

  const clear =
    document.getElementById(
      "smartSearchClear"
    );

  if(!input || !results)
    return;

  const query =
    input.value
      .trim()
      .toLowerCase();

  if(clear){

    clear.classList.toggle(
      "visible",
      query.length > 0
    );

  }

  let items =
    getSmartSearchData();

  if(
    archiveSmartSearchType !==
    "all"
  ){

    items =
      items.filter(
        item =>
          item.type ===
          archiveSmartSearchType
      );

  }

  if(query){

    items =
      items.filter(
        item => {

          const searchable =
            [
              item.title,
              item.meta,
              item.type
            ]
            .filter(Boolean)
            .join(" ")
            .toLowerCase();

          return searchable.includes(
            query
          );

        }
      );

  }

  if(!query){

    results.innerHTML = `

      <div
        class="smart-empty">

        <div
          class="smart-empty-icon">
          🔎
        </div>

        <strong>
          Search the archive
        </strong>

        <div
          style="
            margin-top:5px;
            font-size:12px;
          ">
          Find activities, members,
          ambassadors and magazines.
        </div>

      </div>

    `;

    return;

  }

  if(!items.length){

    results.innerHTML = `

      <div
        class="smart-empty">

        <div
          class="smart-empty-icon">
          🫧
        </div>

        <strong>
          No results found
        </strong>

        <div
          style="
            margin-top:5px;
            font-size:12px;
          ">
          Try another search.
        </div>

      </div>

    `;

    return;

  }

  results.innerHTML =
    items
      .slice(0,50)
      .map(
        item => {

          const image =
            item.image
            ? `
              <img
                src="${escapeHtml(
                  item.image
                )}"
                alt=""
                class="
                  smart-result-image
                ">
            `
            : `
              <div
                class="
                  smart-result-image
                  smart-category-${item.type}
                ">
                ${item.emoji}
              </div>
            `;

          return `

            <button
              class="smart-result"
              onclick="
                executeSmartResult(
                  '${escapeHtml(
                    item.id
                  )}'
                )
              ">

              ${image}

              <span
                class="smart-result-info">

                <span
                  class="smart-result-title">
                  ${escapeHtml(
                    item.title
                  )}
                </span>

                <span
                  class="smart-result-meta">
                  ${escapeHtml(
                    item.meta ||
                    item.type
                  )}
                </span>

              </span>

              <span
                class="
                  smart-result-arrow
                ">
                ›
              </span>

            </button>

          `;

        }
      )
      .join("");

}

              ) ||
              event.target.closest(
                "a"
              ) ||
              event.target.closest(
                "input"
              )

                  activityTitle

                activity
              );

          }
        );

    );
